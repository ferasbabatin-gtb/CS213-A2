enum ExecutionStatus {
    CONTINUE
    HALTED
    INVALID_INSTRUCTION
    LOAD_ERROR
    MEMORY_ERROR
}

class ByteMemory {
    - data : int
    + ByteMemory()
    + ByteMemory(value : int)
    + getData() : int
    + setData(value : int) : void
}

class VoleFloat {
    + {static} toFloat(byteValue : int) : double
    + {static} fromFloat(value : double) : int
}

class Memory {
    - cells : ByteMemory[256]
    + Memory()
    + read(address : int) : int
    + write(address : int, value : int) : void
    + isValidAddress(address : int) : bool
}

class CPURegisters {
    - registers : ByteMemory[16]
    + CPURegisters()
    + read(index : int) : int
    + write(index : int, value : int) : void
    + isValidRegister(index : int) : bool
}

class ExecutionResult {
    - status : ExecutionStatus
    - hasScreenWrite : bool
    - screenValue : int
    + ExecutionResult()
    + ExecutionResult(status : ExecutionStatus, hasScreenWrite : bool, screenValue : int)
    + getStatus() : ExecutionStatus
    + hasOutput() : bool
    + getScreenValue() : int
}

class MachineSnapshot {
    - pc : int
    - ir : int
    - registerValues : int[16]
    - memoryValues : int[256]
    - screenBytes : vector<int>
    - halted : bool
    + getPC() : int
    + getIR() : int
    + getRegisters() : int[16]
    + getMemory() : int[256]
    + getScreenBytes() : vector<int>
    + isHalted() : bool
}

    - mem : Memory&

    - ir : int
    - pc : int
    - halted : bool

    - programEndExclusive : int

    - store(r : int, xy : int) : void
    - move(r : int, s : int) : void
    - add(r : int, s : int, t : int) : void
    - addFloat(r : int, s : int, t : int) : void
    - opOr(r : int, s : int, t : int) : void
    - opAnd(r : int, s : int, t : int) : void
    - opXor(r : int, s : int, t : int) : void
    - rotate(r : int, x : int) : void
    - jump(r : int, xy : int) : ExecutionStatus
    - halt() : void

    + reset(start : int, endExclusive : int) : void
    + setRegister(index : int, value : int) : bool
    + getRegister(index : int) : int
    + copyRegisters() : int[16]
    + getPC() : int
    + getIR() : int
    + isHalted() : bool
}

VoleMachine ..> MachineSnapshot : creates
Menu ..> MachineSnapshot : reads
