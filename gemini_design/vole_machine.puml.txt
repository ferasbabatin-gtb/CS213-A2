class ByteMemory {
    - data: uint8_t
    + getData(): uint8_t
    + setData(value: uint8_t): void
    + toFloat(): float
}

class Memory {
    - cells: ByteMemory[256]
    - onScreenWrite: function
    + read(address: int): uint8_t
    + write(address: int, value: uint8_t): void
    + getCell(address: int): ByteMemory&
    + setScreenWriteCallback(callback: function): void
    + reset(): void
}

class RegisterFile {
    - registers: ByteMemory[16]
    + read(regIndex: int): uint8_t
    + write(regIndex: int, value: uint8_t): void
    + getRegister(regIndex: int): ByteMemory&
    + reset(): void
}

abstract class Instruction {
    + {static} create(rawInstruction: uint16_t): Instruction*
    + {abstract} execute(cpu: CPU&): void
}

class LoadInstruction {
    - reg: uint8_t
    - operand: uint16_t
    - isImmediate: bool
    + LoadInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class StoreInstruction {
    - reg: uint8_t
    - address: uint8_t
    + StoreInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class MoveInstruction {
    - destReg: uint8_t
    - srcReg: uint8_t
    + MoveInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class AddInstruction {
    - destReg: uint8_t
    - regS: uint8_t
    - regT: uint8_t
    - isFloat: bool
    + AddInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class BitwiseInstruction {
    - destReg: uint8_t
    - regS: uint8_t
    - regT: uint8_t
    - operationType: char
    + BitwiseInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class RotateInstruction {
    - reg: uint8_t
    - steps: uint8_t
    + RotateInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class JumpInstruction {
    - reg: uint8_t
    - targetAddress: uint8_t
    + JumpInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class HaltInstruction {
    + HaltInstruction(rawInstruction: uint16_t)
    + execute(cpu: CPU&): void
}

class CPU {
    - memory: Memory
    - registers: RegisterFile
    - pc: int
    - ir: uint16_t
    + fetch(): void
    + decode(): Instruction*
    + executeStep(): bool
    + validateOpcode(opcode: uint8_t): bool
    + isValidInstruction(rawInstruction: uint16_t): bool
    + reset(): void
    + getMemory(): Memory&
    + getRegisters(): RegisterFile&
    + getPC(): int
    + setPC(value: int): void
    + getIR(): uint16_t
}

class VoleMachine {
    - cpu: CPU
    - isRunning: bool
    - programLoaded: bool
    - screenOutput: string
    + loadProgram(filename: string, startAddress: int): bool
    + doOneLine(): bool
    + doAll(): void
    + reset(): void
    + overwriteMemory(address: int, value: uint8_t): void
    + overwriteRegister(regIndex: int, value: uint8_t): void
    + setupScreenCallback(): void
    + appendScreenOutput(value: uint8_t): void
    + getScreenOutput(): string
    + isMachineRunning(): bool
    + getPC(): int
    + getIR(): uint16_t
    + getRegisters(): RegisterFile&
    + getMemory(): Memory&
}

class ConsoleUI {
    - machine: VoleMachine
    + run(): void
    + exit(): void
    + printStatus(): void
    + printScreen(): void
    + printMemoryMatrix(): void
    + printRegisters(): void
}
